# 数据结构的三要素
## 逻辑结构（数据中各元素之间的关系）
- 集合（数据间没有什么联系）
- 线性结构（数据之间存在一对一的关系）
- 树形结构（数据之间是一对多的关系）
- 图结构（数据之间是多对多的关系）
## 存储结构（数据在计算机中的存放）
- 顺序存储（逻辑上面相邻的元素存储在物理上也相邻的存储单元）
- 链式存储（通过指针来储存前一个或者下一个数据的地址，逻辑相邻的元素物理位置不一定相邻）

**数组采用静态申请方式**  ----静态顺序表顺序表一般将第一个元素放在下标为1的上面
```c
#include <iostream>
 const int N=100000;
using namespace std;
//静态封装顺序表
class SqList
{
   int a[N];
   int n;//元素个数
   public: 
   SqList(){
    n=0;
   }
   //顺序表的尾插法
   void push_back(int x){
    a[++n]=x;
   }
   //顺序表的头插法
   void push_front(int x){
      for(int i=n;i>=0;i--){
        a[i+1]=a[i];
      }
      a[1]=x;
      n++;
   }
   //在任意位置插入元素
   void insert(int p,int x){
    for(int i=n;i>=p;i--){
        a[i+1]=a[i];
    }
    a[p]=x;
    n++;
   }
   //删除表尾的元素
   void pop_back(){
    n--;
   }
   //删除表头的元素
   void pop_front(){
    for(int i=2;i<=n;i++){
        a[i-1]=a[i];
    }
    n--;
   }
   //删除任意位置的元素
   void erase(int p){
    for(int i=p+1;i<=n;i++){
        a[i-1]=a[i];
    }
    n--;
   }
   //查找元素-按值查找
   int find(int x){
    for(int i=0;i<n;i++){
        if(a[i]==x){
            return i;
        }
    }
    return -1;
   }
   //查找元素-按下标查找
   int get(int p){
    return a[p];
   }
   //打印顺序表
   void print( ){
    for(int i=0;i<=n;i++){
        cout<<a[i]<<' ';
    }
    cout<<endl<<endl;
   }
};
```

## 树（n个节点构成的有限集合）
- n=0被称为空树
- 有且只有一个根节点
- 其余节点可分为m个互不相交的子集，每个子集也是树，根节点的子树
### 树的基本术语
- 节点间的关系
双亲与孩子   兄弟    祖先   子孙
- 树的高度=树的深度=树的层数
- 节点的深度：他所在的层次
- 结点的高度：以他为根的子树的层数
- 节点的度：孩子的数（度为零的节点叫做叶结点，大于零的结点叫分支结点）
- 树的度：树中结点的最大度数
度为m的数：至少有一个节点的度=m
m叉数：所有的结点的度可以都小于m，甚至可以是空树
- 有序树:各结点的子树从左到右有序不能互换
- 森林：m棵互不相交树的集合

### 孩子表示法
将每个节点的孩子信息存储下来，无根树中将与该节点有关的所有点存储下来
### BTS
BST = 左子树所有值 < 根节点 < 右子树所有值
![在这里插入图片描述](https://i-blog.csdnimg.cn/direct/729d1590777e4ab9b4aa03ad3ac579af.png)
```c
#include <iostream>
using namespace std;
struct TreeNode{
    int val;
    TreeNode *left;
    TreeNode *right;
    TreeNode(int x):val(x),left(nullptr),right(nullptr){}
};
//前序遍历
void front(TreeNode *root){
    if(!root)return ;
    cout<<root->val<<' ';//根
    front(root->left); //左
    front(root->right); //右
}
//中序遍历
void middle(TreeNode *root){
    if(!root)return ;
    middle(root->left);
    cout<<root->val<<' ';
    middle(root->right);
}
//后序遍历
void behind(TreeNode *root){
    if(!root) return ;
    behind(root->left);
    behind(root->right);
    cout<<root->val<<' ';
}
//求树的最大深度
int maxDepth(TreeNode* root) {
       if(root==nullptr)return 0;
    int left=maxDepth(root->left);
    int right=maxDepth(root->right);
    return max(left,right)+1;
    }
//求树的结点
int nodes(TreeNode *root){
    if(!root) return 0;
    int left=nodes(root->left);
    int right=nodes(root->right);
    return (left+right+1);
}
//判断两棵树是否相同
bool issame(TreeNode *p,TreeNode *q){
    if(!p&&!q)return true;
    if(!p||!q)return false;
    return issame(p->left,q->left)&&issame(p->right,q->right);
    
}
//BTS中的搜索
TreeNode* searchBST(TreeNode* root, int val) {
        if(root==nullptr||root->val==val){
            return root;
        }
        if(root->val>val){
           return searchBST(root->left,val);
        }else{
            return searchBST(root->right,val);
        }
    }
//BTS中插入
 TreeNode* insertIntoBST(TreeNode* root, int val) {
        if(!root){
            return new TreeNode(val);
        }
        if(val<root->val){
         root->left=insertIntoBST(root->left,val);
        }else{
            root->right=insertIntoBST(root->right,val);
        }
        return root;
    }
    //BST删除结点
TreeNode *deletenode(TreeNode*root,int key){
   if(!root)return nullptr;
   if(key<root->val){
    deletenode(root->left,key);
   }else if(key>root->val){
    deletenode(root->right,key);
   }else{
    //只有两个或者一个孩子
    if(root->left==nullptr){
        TreeNode * rightroot=root->right;
        delete(root);
        return rightroot;
    }
   }if(root->right=nullptr){
        TreeNode * leftroot=root->left;
        delete(root);
        return leftroot;
    }else{
    //有两个孩子
    TreeNode *objroot=root->right;
    while(objroot->left!=nullptr){
        objroot=objroot->left;
    }
    root->val=objroot->val;
    root->right=deletenode(root->right,objroot->val);
    return root;
}
}
int main(){
    TreeNode *root=new TreeNode (1);
    root->left=new TreeNode(2);
    root->right=new TreeNode(3);
    root->left->left=new TreeNode(4);
    root->left->right=new TreeNode(5);
    root->right->left=new TreeNode(6);
    root->right->right=new TreeNode(7);
    front(root);
    cout<<endl;
    middle(root);
    cout<<endl;
    behind(root);
    cout<<endl;
    cout<<maxDepth(root)<<endl;
    cout<<nodes(root)<<endl;
}
```
## 堆（能快速找到最大值/最小值的结构）
- 必须是完全二叉树
- 大根堆（父结点的值大于等于子结点）
- 小根堆（父结点的值小于等于子结点）
```c
#include <iostream>

}
```
